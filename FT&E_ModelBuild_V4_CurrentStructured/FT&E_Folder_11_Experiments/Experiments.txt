B. Experiments Conducted\par
1. Supply Chain Resilience (Synthetic Data)\par
Objective: Measure how forgiveness thresholds mitigate disruptions in a simple six-node supply chain.\par
Methodology:\par
Setup: Defined nodes representing suppliers, distributors, and retailers within a directed graph.\par
Simulation of Disruptions: Applied disruptions based on excessive lead times and high defect rates, exceeding predefined thresholds.\par
Forgiveness Application: Implemented forgiveness thresholds probabilistically to resolve disruptions, allowing for selective mitigation.\par
Code:\par
 import networkx as nx\par
import random\par
\par
def simulate_supply_chain(forgiveness, disruption_rate=0.3):\par
    # Define the nodes representing suppliers, distributors, and retailers\par
    nodes = ["Supplier1", "Supplier2", "Distributor1", "Distributor2", "Retailer1", "Retailer2"]\par
    edges = [("Supplier1", "Distributor1"), ("Supplier2", "Distributor2"), \par
             ("Distributor1", "Retailer1"), ("Distributor2", "Retailer2")]\par
    G = nx.DiGraph()\par
    G.add_edges_from(edges)\par
\par
    # Simulate disruptions based on the disruption rate\par
    disrupted_edges = \{edge: random.random() < disruption_rate for edge in edges\}\par
    for edge, disrupted in disrupted_edges.items():\par
        if disrupted:\par
            # Apply forgiveness threshold to potentially resolve the disruption\par
            if random.random() < forgiveness:\par
                disrupted_edges[edge] = False\par
\par
    # Calculate the recovery rate based on unresolved disruptions\par
    total_disruptions = sum(disrupted_edges.values())\par
    recovery_rate = 1 - (total_disruptions / len(edges))\par
    return recovery_rate\par
\par
Findings:\par
20% Forgiveness: Recovery Rate = 24.51%.\par
50% Forgiveness: Recovery Rate = 48.04%.\par
80% Forgiveness: Recovery Rate = 79.41%.\par
Insight: Forgiveness thresholds significantly enhance resilience by increasing the recovery rate proportionally with higher forgiveness levels.\par

 import numpy as np\par
from sklearn.neural_network import MLPClassifier\par
from sklearn.metrics import accuracy_score\par

def simulate_neural_network(forgiveness, data_size=1000, feature_count=20):\par
    # Generate synthetic feature data\par
    X = np.random.rand(data_size, feature_count)\par
    y = np.random.choice([0, 1], size=data_size)\par
    \par
    # Initialize neural network with adjusted learning rate based on forgiveness\par
    model = MLPClassifier(hidden_layer_sizes=(50,), max_iter=500, \par
                          learning_rate_init=0.001 * (1 + forgiveness))\par
    model.fit(X, y)\par
    accuracy = model.score(X, y)\par
    return accuracy\par

import numpy as np\par
import matplotlib.pyplot as plt\par
\par
def cosine_similarity(vec1, vec2):\par
    return np.dot(vec1, vec2) / (np.linalg.norm(vec1) * np.linalg.norm(vec2))\par
\par
def feedback_strength(t, lambda_):\par
    return np.exp(-lambda_ * t)\par
\par
# Parameters\par
H = np.array([1.0, 0.5])  # Initial human intent\par
A = np.array([0.5, 1.0])  # Initial AI response\par
alpha = 0.1  # Learning rate for human intent\par
gamma = 0.05  # Adaptation rate for AI response\par
lambda_ = 0.01  # Feedback decay rate\par
iterations = 100\par
\par
# Tracking metrics\par
alignment_scores = []\par
feedback_strengths = []\par
\par
for t in range(iterations):\par
    # Calculate current feedback strength\par
    feedback = feedback_strength(t, lambda_)\par
    feedback_strengths.append(feedback)\par
\par
    # Update alignment score\par
    alignment = cosine_similarity(H, A)\par
    alignment_scores.append(alignment)\par
\par
    # Update human intent\par
    H = H + alpha * (A - H) * feedback\par
\par
    # Update AI response\par
    A = A + gamma * (H - A) * feedback\par
\par
# Plot results\par
plt.figure(figsize=(12, 6))\par
\par
# Plot alignment score\par
plt.subplot(1, 2, 1)\par
plt.plot(range(iterations), alignment_scores, label="Alignment Score")\par
plt.xlabel("Iteration")\par
plt.ylabel("Cosine Similarity")\par
plt.title("Alignment Improvement Over Iterations")\par
plt.legend()\par
\par
# Plot feedback strength\par
plt.subplot(1, 2, 2)\par
plt.plot(range(iterations), feedback_strengths, label="Feedback Strength")\par
plt.xlabel("Iteration")\par

plt.title("Feedback Strength Decay")\par

# Import necessary libraries for the refined simulation\par
import numpy as np\par
import matplotlib.pyplot as plt\par
\par
# Define parameters for the simulation\par
iterations = 100\par
learning_rate_human = 0.05  # Learning rate for human intent updates\par
learning_rate_ai = 0.05  # Learning rate for AI response updates\par
feedback_decay = 0.01  # Decay rate for feedback loop strength\par
initial_human_intent = 0.5  # Starting human intent alignment (scale: 0 to 1)\par
initial_ai_response = 0.5  # Starting AI response alignment (scale: 0 to 1)\par
\par
# Initialize arrays to store alignment over iterations\par
human_intent_history = [initial_human_intent]\par
ai_response_history = [initial_ai_response]\par
feedback_strength_history = [1.0]  # Initial feedback strength\par
\par
# Iterative refinement process\par
human_intent = initial_human_intent\par
ai_response = initial_ai_response\par
feedback_strength = 1.0\par
\par
for i in range(iterations):\par
    # Update AI response based on current human intent and feedback strength\par
    ai_response += learning_rate_ai * feedback_strength * (human_intent - ai_response)\par
    ai_response = np.clip(ai_response, 0, 1)  # Ensure response stays within [0, 1]\par
\par
    # Update human intent based on evaluation of AI response and feedback strength\par
    human_intent += learning_rate_human * feedback_strength * (ai_response - human_intent)\par
    human_intent = np.clip(human_intent, 0, 1)  # Ensure intent stays within [0, 1]\par
\par
    # Decay feedback strength over time\par
    feedback_strength *= (1 - feedback_decay)\par
\par
    # Record the values for this iteration\par
    human_intent_history.append(human_intent)\par
    ai_response_history.append(ai_response)\par
    feedback_strength_history.append(feedback_strength)\par
\par
# Plot results to visualize the refinement process\par
plt.figure(figsize=(10, 6))\par
plt.plot(human_intent_history, label="Human Intent")\par
plt.plot(ai_response_history, label="AI Response")\par

plt.xlabel("Iteration")\par
plt.ylabel("Alignment / Feedback Strength")\par
plt.legend()\par
plt.grid()\par
plt.show()\par
